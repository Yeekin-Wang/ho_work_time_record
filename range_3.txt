1312:     let startMinute: number = punches[0];
1313:     let endMinute: number = punches[punches.length - 1];

1315:       return 0;

1318:     let index: number = 0;
1319:     while (index < this.excludedRanges.length) {
1320:       let range: ExcludedRange = this.excludedRanges[index];

1329:       return 0;
